document.getElementById("form-login").addEventListener("submit", async (e) => { e.preventDefault(); const email = document.getElementById("email-input").value; const senha = document.getElementById("senha-input").value; const nomeProfissional = document.getElementById("nome-profissional")?.value || ""; const corenProfissional = document.getElementById("coren-profissional")?.value || ""; if (typeof supabase === 'undefined') { alert("Erro: Cliente Supabase não carregado."); return; } try { if (modoAtual === "cadastro") { // 1. Cria a conta no Auth do Supabase const { data: authData, error: authError } = await supabase.auth.signUp({ email, senha }); if (authError) throw authError; const userId = authData.user?.id; // 2. Salva o perfil na tabela (Se for profissional, nasce com aprovado: false) if (userId) { const isProfissional = (perfilAtual === 'profissional'); const { error: dbError } = await supabase.from('perfis_usuarios').insert([ { id: userId, email: email, tipo_perfil: perfilAtual, nome: isProfissional ? nomeProfissional : 'Paciente', coren: isProfissional ? corenProfissional : null, aprovado: !isProfissional // Pacientes aprovados direto; profissionais exigem moderação } ]); if (dbError) console.warn("Erro ao salvar perfil:", dbError.message); } if (perfilAtual === 'profissional') { alert("Cadastro realizado com sucesso! Sua conta de profissional está pendente de aprovação pela coordenação da SESAPI/HGV."); window.location.href = "login.html"; } else { alert("Cadastro de paciente realizado com sucesso!"); window.location.href = "dashboard.html"; } } else { // 1. Faz o login const { data: authData, error: authError } = await supabase.auth.signInWithPassword({ email, senha }); if (authError) throw authError; const userId = authData.user?.id; // 2. Consulta o perfil e o status de aprovação no banco const { data: perfilData, error: perfilError } = await supabase .from('perfis_usuarios') .select('tipo_perfil, aprovado') .eq('id', userId) .single(); if (perfilData?.tipo_perfil === 'profissional') { if (!perfilData.aprovado) { alert("Acesso restrito: Sua conta de profissional ainda não foi aprovada pela coordenação."); await supabase.auth.signOut(); // Desloga o usuário return; } alert("Login de profissional efetuado com sucesso!"); window.location.href = "profissional.html"; } else { alert("Login efetuado com sucesso!"); window.location.href = "dashboard.html"; } } } catch (err) { alert("Erro na autenticação: " + err.message); } });
